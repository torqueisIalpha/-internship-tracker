import java.time.LocalDate; import java.time.temporal.ChronoUnit; import java.util.ArrayList; import java.util.List; import java.util.Scanner; public class Main { static class Application { String company; String role; LocalDate deadline; String status; String memo; Application(String company, String role, LocalDate deadline, String status, String memo) { this.company = company; this.role = role; this.deadline = deadline; this.status = status; this.memo = memo; } } static List apps = new ArrayList<>(); static Scanner scanner = new Scanner(System.in); public static void main(String[] args) { // Pre-loaded sample data matching your tracker apps.add(new Application("Orbit Analytic", "ML research intern", LocalDate.now().minusDays(1), "Interview", "Email recruiter for an extension")); apps.add(new Application("Northwind Labs", "Data analyst intern", LocalDate.now().plusDays(2), "Interview", "Prepare SQL examples")); apps.add(new Application("Bluepeak Systems", "Frontend intern", LocalDate.now().plusDays(6), "Test", "Complete coding challenge")); apps.add(new Application("Greenleaf Health", "Product intern", LocalDate.now().plusDays(12), "Interview", "")); while (true) { printDashboard(); System.out.println("\nChoose an option:"); System.out.println("1. Add application"); System.out.println("2. Update status"); System.out.println("3. Delete application"); System.out.println("4. Filter by status"); System.out.println("5. Exit"); System.out.print("> "); String choice = scanner.nextLine().trim(); switch (choice) { case "1": addApplication(); break; case "2": updateStatus(); break; case "3": deleteApplication(); break; case "4": filterByStatus(); break; case "5": System.out.println("Exiting Internship Tracker."); return; default: System.out.println("Invalid choice. Try again."); } } } static void printDashboard() { System.out.println("\n=========================================="); System.out.println(" INTERNSHIP TRACKER "); System.out.println("=========================================="); long needsAction = 0; for (Application app : apps) { long days = ChronoUnit.DAYS.between(LocalDate.now(), app.deadline); if (!app.status.equals("Offer") && !app.status.equals("Rejected") && days <= 3) { needsAction++; } } if (needsAction > 0) { System.out.println("[!] ALERT: " + needsAction + " application(s) need action (due within 3 days or overdue)."); } System.out.println("\n--- All Applications (" + apps.size() + ") ---"); for (int i = 0; i < apps.size(); i++) { Application app = apps.get(i); long days = ChronoUnit.DAYS.between(LocalDate.now(), app.deadline); String dueText = formatDue(days); System.out.printf("%d. %s - %s [%s] | Deadline: %s (%s)\n", i + 1, app.company, app.role, app.status, app.deadline, dueText); if (!app.memo.isEmpty()) { System.out.println(" Next action: " + app.memo); } } } static String formatDue(long days) { if (days < 0) return "Overdue by " + Math.abs(days) + " day(s)"; if (days == 0) return "Due today"; return "Due in " + days + " day(s)"; } static void addApplication() { System.out.print("Company: "); String company = scanner.nextLine().trim(); System.out.print("Role: "); String role = scanner.nextLine().trim(); System.out.print("Deadline (YYYY-MM-DD): "); LocalDate deadline; try { deadline = LocalDate.parse(scanner.nextLine().trim()); } catch (Exception e) { System.out.println("Invalid date format. Using today's date."); deadline = LocalDate.now(); } System.out.print("Status (Applied / Test / Interview / Offer / Rejected): "); String status = scanner.nextLine().trim(); System.out.print("Next action (optional): "); String memo = scanner.nextLine().trim(); apps.add(new Application(company, role, deadline, status, memo)); System.out.println("Application added."); } static void updateStatus() { System.out.print("Enter application number to update: "); try { int index = Integer.parseInt(scanner.nextLine().trim()) - 1; if (index >= 0 && index < apps.size()) { System.out.print("New status (Applied / Test / Interview / Offer / Rejected): "); apps.get(index).status = scanner.nextLine().trim(); System.out.println("Status updated."); } else { System.out.println("Invalid number."); } } catch (NumberFormatException e) { System.out.println("Please enter a valid number."); } } static void deleteApplication() { System.out.print("Enter application number to delete: "); try { int index = Integer.parseInt(scanner.nextLine().trim()) - 1; if (index >= 0 && index < apps.size()) { apps.remove(index); System.out.println("Application deleted."); } else { System.out.println("Invalid number."); } } catch (NumberFormatException e) { System.out.println("Please enter a valid number."); } } static void filterByStatus() { System.out.print("Enter status to filter by (Applied / Test / Interview / Offer / Rejected): "); String target = scanner.nextLine().trim(); System.out.println("\n--- Filtered: " + target + " ---"); for (Application app : apps) { if (app.status.equalsIgnoreCase(target)) { System.out.println("- " + app.company + " (" + app.role + ") | Deadline: " + app.deadline); } } } }
